class ResumeData(BaseModel):
    templateId: str = \"modern-minimal\"
    personal: PersonalInfo = Field(default_factory=PersonalInfo)
    experience: List[Experience] = Field(default_factory=list)
    education: List[Education] = Field(default_factory=list)
    skills: List[str] = Field(default_factory=list)
    projects: List[Project] = Field(default_factory=list)
    certifications: List[Certification] = Field(default_factory=list)

class Resume(BaseModel):
    model_config = ConfigDict(extra=\"ignore\")

    id: str = Field(default_factory=lambda: str(uuid.uuid4()))
    share_id: str = Field(default_factory=lambda: uuid.uuid4().hex[:10])
    data: ResumeData = Field(default_factory=ResumeData)
    created_at: str = Field(default_factory=lambda: datetime.now(timezone.utc).isoformat())
    updated_at: str = Field(default_factory=lambda: datetime.now(timezone.utc).isoformat())


class ResumeCreate(BaseModel):
    data: ResumeData


class ResumeUpdate(BaseModel):
    data: ResumeData


# ---------- Routes ----------
@api_router.get(\"/\")
async def root():
    return {\"message\": \"Resumely API\", \"status\": \"ok\"}


@api_router.post(\"/resumes\", response_model=Resume)
async def create_resume(payload: ResumeCreate):
    resume = Resume(data=payload.data)
    doc = resume.model_dump()
    await db.resumes.insert_one(doc)
    return resume


@api_router.get(\"/resumes/{share_id}\", response_model=Resume)
async def get_resume(share_id: str):
    doc = await db.resumes.find_one({\"share_id\": share_id}, {\"_id\": 0})
    if not doc:
        raise HTTPException(status_code=404, detail=\"Resume not found\")
    return Resume(**doc)


@api_router.put(\"/resumes/{share_id}\", response_model=Resume)
async def update_resume(share_id: str, payload: ResumeUpdate):
    now = datetime.now(timezone.utc).isoformat()
    result = await db.resumes.find_one_and_update(
        {\"share_id\": share_id},
        {\"$set\": {\"data\": payload.data.model_dump(), \"updated_at\": now}},
        return_document=True,
        projection={\"_id\": 0},
    )
    if not result:
        raise HTTPException(status_code=404, detail=\"Resume not found\")
    return Resume(**result)
