"import React, { useEffect, useState } from \"react\";
import { useParams, Link } from \"react-router-dom\";
import Nav from \"@/components/Nav\";
import { renderTemplate } from \"@/templates/registry\";
import { getResume } from \"@/lib/api\";
import { Button } from \"@/components/ui/button\";
import { Download, Loader2 } from \"lucide-react\";

export default function SharedView() {
  const { shareId } = useParams();
  const [state, setState] = useState({ loading: true, error: null, data: null });

  useEffect(() => {
    let mounted = true;
    getResume(shareId)
      .then((r) => mounted && setState({ loading: false, error: null, data: r.data }))
      .catch((e) => mounted && setState({ loading: false, error: \"Resume not found\", data: null }));
    return () => { mounted = false; };
  }, [shareId]);

  return (
    <div className=\"min-h-screen bg-white\">
      <Nav
        right={
          <div className=\"flex items-center gap-2 no-print\">
            <Link
              to=\"/templates\"
              data-testid=\"shared-cta\"
              className=\"text-sm border border-zinc-300 px-3 py-2 rounded-sm hover:bg-zinc-50\"
            >
              Build your own
            </Link>
            <Button data-testid=\"shared-download-pdf\" className=\"rounded-sm bg-black text-white hover:bg-zinc-800\" onClick={() => window.print()}>
              <Download size={14} className=\"mr-1.5\" /> PDF
            </Button>
          </div>
        }
      />

      <div className=\"bg-zinc-100 min-h-[calc(100vh-4rem)] flex justify-center py-10 px-4\">
        {state.loading && (
          <div className=\"text-zinc-500 flex items-center gap-2 mt-24\" data-testid=\"shared-loading\">
            <Loader2 size={16} className=\"animate-spin\" /> Loading resume...
          </div>
        )}
        {state.error && (
          <div className=\"text-center mt-24\" data-testid=\"shared-error\">
            <h2 className=\"font-heading text-3xl font-black tracking-tight\">404</h2>
            <p className=\"mt-2 text-zinc-500\">{state.error}</p>
            <Link to=\"/\" className=\"mt-4 inline-block text-sm underline\">Back home</Link>
          </div>
        )}
        {state.data && (
          <div className=\"print-target\">{renderTemplate(state.data.templateId, state.data)}</div>
        )}
      </div>
    </div>
  );
}
"
