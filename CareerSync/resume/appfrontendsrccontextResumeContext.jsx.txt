"import React, { createContext, useContext, useState, useCallback } from \"react\";
import { sampleResume } from \"@/data/sampleResume\";

const ResumeContext = createContext(null);

export const ResumeProvider = ({ children, initial }) => {
  const [resume, setResume] = useState(initial || sampleResume);
  const [shareId, setShareId] = useState(null);

  const update = useCallback((patch) => {
    setResume((r) => ({ ...r, ...(typeof patch === \"function\" ? patch(r) : patch) }));
  }, []);

  const updateField = useCallback((section, index, key, value) => {
    setResume((r) => {
      const list = [...(r[section] || [])];
      list[index] = { ...list[index], [key]: value };
      return { ...r, [section]: list };
    });
  }, []);

  const addItem = useCallback((section, item) => {
    setResume((r) => ({ ...r, [section]: [...(r[section] || []), item] }));
  }, []);

  const removeItem = useCallback((section, index) => {
    setResume((r) => ({ ...r, [section]: r[section].filter((_, i) => i !== index) }));
  }, []);

  return (
    <ResumeContext.Provider value={{ resume, setResume, update, updateField, addItem, removeItem, shareId, setShareId }}>
      {children}
    </ResumeContext.Provider>
  );
};

export const useResume = () => {
  const ctx = useContext(ResumeContext);
  if (!ctx) throw new Error(\"useResume must be used inside ResumeProvider\");
  return ctx;
};
